│  function formatContent(content) {                                       │
│    let html = '';                                                        │
│    const lines = content.split('\n');                                    │
│    let inCode = false;                                                   │
│    let codeBuffer = [];                                                  │
│                                                                          │
│    for (let line of lines) {                                             │
│      if (line.trim().startsWith('```')) {                                │
│        if (inCode) {                                                     │
│          html +=                                                         │
│  `<pre><code>${escapeHtml(codeBuffer.join('\n'))}</code></pre>`;         │
│          codeBuffer = [];                                                │
│        }                                                                 │
│        inCode = !inCode;                                                 │
│        continue;                                                         │
│      }                                                                   │
│      if (inCode) {                                                       │
│        codeBuffer.push(line);                                            │
│        continue;                                                         │
│      }                                                                   │
│                                                                          │
│      const trimmed = line.trim();                                        │
│      if (trimmed === '') {                                               │
│        html += '<p>&nbsp;</p>';                                          │
│      } else if (trimmed.startsWith('### ')) {                            │
│        html += `<h3>${escapeHtml(trimmed.slice(4))}</h3>`;               │
│      } else if (trimmed.startsWith('## ')) {                             │
│        html += `<h2>${escapeHtml(trimmed.slice(3))}</h2>`;               │
│      } else if (trimmed.startsWith('- ')) {                              │
│        html += `<li>${escapeHtml(trimmed.slice(2))}</li>`;               │
│      } else if (trimmed.startsWith('> ')) {                              │
│        html +=                                                           │
│  `<blockquote>${escapeHtml(trimmed.slice(2))}</blockquote>`;             │
│      } else {                                                            │
│        const inlineCode = escapeHtml(trimmed).replace(/`([^`]+)`/g,      │
│  '<code>$1</code>');                                                     │
│        html += `<p>${inlineCode}</p>`;                                   │
│      }                                                                   │
│    }                                                                     │
│                                                                          │
│    if (inCode && codeBuffer.length) {                                    │
│      html +=                                                             │
│  `<pre><code>${escapeHtml(codeBuffer.join('\n'))}</code></pre>`;         │
│    }                                                                     │
│                                                                          │

│  }                                                                       │
