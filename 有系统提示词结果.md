│          const STORAGE_KEY = 'taskflow_todos';                           │
│                                                                          │
│          /** 获取所有任务 */                                             │
│          function getTodos() {                                           │
│              try {                                                       │
│                  const raw = localStorage.getItem(STORAGE_KEY);          │
│                  return raw ? JSON.parse(raw) : [];                      │
│              } catch {                                                   │
│                  return [];                                              │
│              }                                                           │
│          }                                                               │
│                                                                          │
│          /** 保存任务列表 */                                             │
│          function saveTodos(todos) {                                     │
│              localStorage.setItem(STORAGE_KEY, JSON.stringify(todos));   │
│          }                                                               │
│                                                                          │
│          //                                                              │
│  ============================================================            │
│          //  状态                                                        │
│          //                                                              │
│  ============================================================            │
│                                                                          │
│          let todos = getTodos();                                         │
│          let currentFilter = 'all'; // 'all' | 'pending' | 'done'        │
│                                                                          │
│          //                                                              │
│  ============================================================            │
│          //  DOM引用                                                     │
│          //                                                              │

│                                                                          │
│          const todoListEl = document.getElementById('todoList');         │

│          const filterTabs = document.querySelectorAll('.filter-tab');    │
│                                                                          │
│          //                                                              │
│  ============================================================            │

│          //                                                              │
│  ============================================================            │
│                                                                          │
│          /** 生成短时间戳 */                                             │
│          function getTimeLabel() {                                       │
│              const now = new Date();                                     │
│              const h = String(now.getHours()).padStart(2, '0');          │
│              const m = String(now.getMinutes()).padStart(2, '0');        │
│              return `${h}:${m}`;                                         │
│          }                                                               │
│                                                                          │
│          /** 获取当前日期（用于显示） */                                 │
│          function getDateLabel() {                                       │
│              const now = new Date();                                     │
│              const month = now.getMonth() + 1;                           │
│              const day = now.getDate();                                  │
│              const weekdays = ['日', '一', '二', '三', '四', '五',       │
│  '六'];                                                                  │
│              const wd = weekdays[now.getDay()];                          │
│              return `${month}月${day}日 周${wd}`;                        │
│          }                                                               │
│                                                                          │
│          /** 生成唯一ID */                                               │
│          function genId() {                                              │
│              return Date.now().toString(36) +                            │
│  Math.random().toString(36).slice(2, 6);                                 │
│          }                                                               │
│                                                                          │
│          //                                                              │
│  ============================================================            │
│          //  渲染引擎                                                    │
│          //                                                              │
│  ============================================================            │
│                                                                          │
│          function render() {                                             │
│              // 1. 筛选                                                  │
│              let filtered = [...todos];                                  │
│              if (currentFilter === 'pending') {                          │
│                  filtered = filtered.filter(t => !t.done);               │
│              } else if (currentFilter === 'done') {                      │
│                  filtered = filtered.filter(t => t.done);                │
│              }                                                           │
│                                                                          │
│              // 2.                                                       │
│  排序：未完成的在上，按创建时间倒序；已完成的在下，按完成时间倒序        │
│              filtered.sort((a, b) => {                                   │
│                  if (a.done !== b.done) return a.done ? 1 : -1;          │
│                  return b.createdAt - a.createdAt;                       │
│              });                                                         │
│                                                                          │
│              // 3. 更新统计                                              │
│              const total = todos.length;                                 │
│              const done = todos.filter(t => t.done).length;              │
│              const pending = total - done;                               │
│              totalCount.textContent = total;                             │
│              doneCount.textContent = done;                               │
│              pendingCount.textContent = pending;                         │
│                                                                          │
│              // 4. 渲染列表                                              │
│              if (filtered.length === 0) {                                │
│                  todoListEl.innerHTML = '';                              │
│                  emptyStateEl.style.display = 'block';                   │
│                  // 根据不同筛选显示不同文案                             │
│                  if (total === 0) {                                      │
│                      emptyTextEl.textContent = '还没有任何任务';         │
│                  } else if (currentFilter === 'pending') {               │
│                      emptyTextEl.textContent =                           │
│  '太棒了，所有任务都已完成！ 🎉';                                        │
│                  } else if (currentFilter === 'done') {                  │
│                      emptyTextEl.textContent = '还没有已完成的任务';     │
│                  }                                                       │
│                  return;                                                 │
│              }                                                           │
│                                                                          │
│              emptyStateEl.style.display = 'none';                        │
│                                                                          │
│              let html = '';                                              │
│              for (const todo of filtered) {                              │
│                  const checkedClass = todo.done ? 'checked' : '';        │
│                  const textClass = todo.done ? 'done' : '';              │
│                  html += `                                               │
│                      <li class="todo-item" data-id="${todo.id}">         │
│                          <div class="checkbox ${checkedClass}"           │
│  data-action="toggle"></div>                                             │
│                          <span class="task-text                          │
│  ${textClass}">${escapeHtml(todo.text)}</span>                           │
│                          <button class="delete-btn"                      │
│  data-action="delete" title="删除任务">✕</button>                        │
│                      </li>                                               │
│                  `;                                                      │
│              }                                                           │
│              todoListEl.innerHTML = html;                                │
│          }                                                               │
│                                                                          │
│          /** 简单防XSS */                                                │
│          function escapeHtml(text) {                                     │
│              const div = document.createElement('div');                  │
│              div.textContent = text;                                     │
│              return div.innerHTML;                                       │
│          }                                                               │
│                                                                          │
│          //                                                              │
│  ============================================================            │
│          //  操作函数                                                    │
│          //                                                              │
│  ============================================================            │
│                                                                          │
│          function addTodo(text) {                                        │
│              text = text.trim();                                         │
│              if (!text) return false;                                    │
│                                                                          │
│              const todo = {                                              │
│                  id: genId(),                                            │
│                  text: text,                                             │
│                  done: false,                                            │
│                  createdAt: Date.now(),                                  │
│                  doneAt: null,                                           │
│              };                                                          │
│              todos.unshift(todo);                                        │
│              saveTodos(todos);                                           │
│              render();                                                   │
│              return true;                                                │
│          }                                                               │
│                                                                          │
│          function toggleTodo(id) {                                       │
│              const todo = todos.find(t => t.id === id);                  │
│              if (!todo) return;                                          │
│              todo.done = !todo.done;                                     │
│              todo.doneAt = todo.done ? Date.now() : null;                │
│              saveTodos(todos);                                           │
│              render();                                                   │
│          }                                                               │
│                                                                          │
│          function deleteTodo(id) {                                       │
│              todos = todos.filter(t => t.id !== id);                     │
│              saveTodos(todos);                                           │
│              render();                                                   │
│          }                                                               │
│                                                                          │
│          function clearDone() {                                          │
│              const hadDone = todos.some(t => t.done);                    │
│              if (!hadDone) return;                                       │
│              todos = todos.filter(t => !t.done);                         │
│              saveTodos(todos);                                           │
│              render();                                                   │
│          }                                                               │
│                                                                          │

│  ============================================================            │

│                                                                          │
